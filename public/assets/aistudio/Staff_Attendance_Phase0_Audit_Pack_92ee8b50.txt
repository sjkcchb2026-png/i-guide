rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Default deny all
    match /{document=**} {
      allow read, write: if false;
    }

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn() && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    // Admins registry: Readable by authenticated users; Client writes are STRICTLY PROHIBITED.
    // Admins are created exclusively by the server backend with SETUP_TOKEN.
    match /admins/{userId} {
      allow read: if isSignedIn();
      allow write: if false;
    }

    // School Settings: Readable by public portal and staff; Client writes are STRICTLY PROHIBITED.
    match /school_settings/{settingId} {
      allow read: if true;
      allow write: if false;
    }

    // Staff accounts: Staff can only read their own profile; Principal can list all profiles.
    // Client writes are STRICTLY PROHIBITED (managed solely by server backend with Principal auth).
    match /users/{userId} {
      allow get: if isSignedIn() && (isOwner(userId) || isAdmin());
      allow list: if isAdmin();
      allow write: if false;
    }

    // Attendance Records: Staff can only read their own records; Principal can read all.
    // Client writes are STRICTLY PROHIBITED. All punches are executed by backend transactions.
    match /attendance/{recordId} {
      allow get: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow list: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow write: if false;
    }

    // Isolated GPS Coordinates: Readable EXCLUSIVELY by Principal. Staff and public cannot read.
    // Client writes are STRICTLY PROHIBITED.
    match /attendance_geo/{recordId} {
      allow read: if isAdmin();
      allow write: if false;
    }

    // User Attendance State: Readable by owner or Principal. Client writes prohibited.
    match /user_attendance_state/{userId} {
      allow read: if isSignedIn() && (request.auth.uid == userId || isAdmin());
      allow write: if false;
    }

    // School Calendar & Holidays: Readable by authenticated users; Principal can write.
    match /school_calendar/{calId} {
      allow read: if isSignedIn();
      allow write: if isAdmin();
    }

    // Absence / Leave / Official Duty Records: Readable by owner or Principal; Principal can write.
    match /absence_records/{recordId} {
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin());
      allow write: if isAdmin();
    }
  }
}

  getFirestore, 
  doc, 
  getDocFromServer 
} from 'firebase/firestore';
import firebaseConfig from '../firebase-applet-config.json';

// Initialize primary Firebase App
export const app = initializeApp(firebaseConfig);
export const db = getFirestore(app, firebaseConfig.firestoreDatabaseId);
export const auth = getAuth(app);
export { firebaseConfig };

// Error handling conforming to Firebase skill specs
export enum OperationType {
  CREATE = 'create',
  UPDATE = 'update',
  DELETE = 'delete',
  LIST = 'list',
  GET = 'get',
  WRITE = 'write',
}

export interface FirestoreErrorInfo {
  error: string;
  operationType: OperationType;
  path: string | null;
  authInfo: {
    userId?: string | null;
    email?: string | null;
    emailVerified?: boolean | null;
    isAnonymous?: boolean | null;
    tenantId?: string | null;
    providerInfo?: {
      providerId?: string | null;
      email?: string | null;
    }[];
  };
}

export function handleFirestoreError(error: unknown, operationType: OperationType, path: string | null): never {
  const errInfo: FirestoreErrorInfo = {
    error: error instanceof Error ? error.message : String(error),
    authInfo: {
      userId: auth.currentUser?.uid,
      email: auth.currentUser?.email,
      emailVerified: auth.currentUser?.emailVerified,
      isAnonymous: auth.currentUser?.isAnonymous,
      tenantId: auth.currentUser?.tenantId,
      providerInfo: auth.currentUser?.providerData?.map(provider => ({
        providerId: provider.providerId,
        email: provider.email,
      })) || []
    },
    operationType,
    path
  };
  console.error('Firestore Error:', JSON.stringify(errInfo));
  throw new Error(JSON.stringify(errInfo));
}

// Startup connection verification
export async function testFirestoreConnection(): Promise<boolean> {
  try {
    await getDocFromServer(doc(db, 'test', 'connection'));
    return true;
  } catch (error) {
    if (error instanceof Error && error.message.includes('the client is offline')) {
      console.warn("Please check your Firebase configuration or internet connection.");
      return false;
    }
    // "permission-denied" is expected on /test/connection since rules block unknown paths, meaning connection is live
    return true;
  }
}

      } else {
        setSchoolSettings(null);
      }
    }, (error) => {
      console.warn('School settings subscription:', error);
    });

    return () => unsub();
  }, []);

  // 2. Listen to Auth State
  useEffect(() => {
    const unsubscribe = onAuthStateChanged(auth, async (user) => {
      setFirebaseUser(user);
      setAccessDeniedReason(null);

      if (!user) {
        setCurrentUserProfile(null);
        setLoading(false);
        return;
      }

      try {
        // 1. Check if user is registered Admin in /admins/{uid}
        const adminDocRef = doc(db, 'admins', user.uid);
        const adminSnap = await getDoc(adminDocRef);
        const isAdmin = adminSnap.exists();

        // 2. Fetch user profile from /users/{uid}
        const userDocRef = doc(db, 'users', user.uid);
        const userSnap = await getDoc(userDocRef);

        if (isAdmin) {
          // Principal profile
          let profile: UserProfile;
          if (userSnap.exists()) {
            profile = userSnap.data() as UserProfile;
            profile.role = 'PRINCIPAL';
            profile.category = 'PRINCIPAL';
          } else {
            profile = {
              uid: user.uid,
              name: user.displayName || '校长 / Guru Besar',
              email: user.email || '',
              category: 'PRINCIPAL',
              role: 'PRINCIPAL',
              hasClerkAccess: true,
              active: true,
            };
          }
          setCurrentUserProfile(profile);
          setActiveView('principal');
        } else if (userSnap.exists()) {
          const profile = userSnap.data() as UserProfile;

          // Check if active
          if (profile.active === false) {
            setAccessDeniedReason('DEACTIVATED');
            setCurrentUserProfile(null);
          } else {
            // Valid active staff member
            setCurrentUserProfile(profile);
            setActiveView('staff');
          }
        } else {
          // Account exists in Firebase Auth but has NO authorized record in /users created by Principal!
          // Strictly forbid access, remove any auto-create fallback!
          setAccessDeniedReason('NOT_REGISTERED');
          setCurrentUserProfile(null);
        }
      } catch (err) {
        console.error('Error loading user profile:', err);
      } finally {
        setLoading(false);
      }
    });

    return () => unsubscribe();
  }, []);

  const handleLogout = async () => {
    try {
      await signOut(auth);
      setAccessDeniedReason(null);
      setCurrentUserProfile(null);
    } catch (err) {
      console.error('Logout error:', err);
    }
  };

  if (loading) {
    return (
      <div className="min-h-screen bg-slate-950 flex flex-col items-center justify-center text-slate-300">
        <div className="w-12 h-12 rounded-2xl bg-blue-600/20 border border-blue-500/30 flex items-center justify-center text-blue-400 mb-4 animate-pulse">

            <div className="w-16 h-16 rounded-2xl bg-rose-950/60 border border-rose-800/60 text-rose-400 flex items-center justify-center mx-auto mb-4">
              {accessDeniedReason === 'DEACTIVATED' ? (
                <UserX className="w-8 h-8" />
              ) : (
                <ShieldAlert className="w-8 h-8" />
              )}
            </div>

                ? '账号已被停用 (Akaun Dinyahaktifkan)'
                : '未获校方授权建档 (Akaun Belum Disahkan)'}
            </h3>

            <p className="text-xs sm:text-sm text-slate-400 leading-relaxed mb-6">
              {accessDeniedReason === 'DEACTIVATED' ? (
                <>
                  您的教职员账号（{firebaseUser.email}）已被校长暂时停用，目前无法进行打卡或查阅出勤。如需恢复打卡权限，请联系校方管理员。
                </>
              ) : (
                <>
                  当前登录账号（{firebaseUser.email}）尚未由校长在出勤系统后台建立档案。本系统禁止自行公开注册，请联系校长开通账号后方可打卡。
                </>
              )}
            </p>

            <button
              onClick={handleLogout}
              className="w-full py-2.5 px-4 bg-slate-800 hover:bg-slate-700 text-slate-200 text-sm font-medium rounded-xl transition flex items-center justify-center space-x-2"
            >
              <LogOut className="w-4 h-4" />
              <span>退出登录 / Log Keluar</span>
            </button>
          </div>
        </main>
      </div>
    );
  }

  // If not logged in, display Login and Initial Setup view
  if (!firebaseUser || !currentUserProfile) {
    return (
      <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col">
        <Header
          currentUser={null}
          schoolSettings={schoolSettings}
          activeView={activeView}
          setActiveView={setActiveView}
        />
        <main className="flex-1">
          <LoginView
            schoolSettings={schoolSettings}
            onLoginSuccess={() => {}}
          />
        </main>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans">
      
      {/* Top Navigation & School Header */}
      <Header
        currentUser={currentUserProfile}
        schoolSettings={schoolSettings}
        activeView={activeView}
        setActiveView={setActiveView}
      />

      {/* Main App Content View */}
      <main className="flex-1 pb-12">
        {activeView === 'principal' && currentUserProfile.role === 'PRINCIPAL' ? (
          <PrincipalDashboard
            currentUser={currentUserProfile}
            schoolSettings={schoolSettings}
            onUpdateSchoolSettings={(updated) => setSchoolSettings(updated)}
          />
        ) : (
          <StaffPunchIn currentUser={currentUserProfile} />
        )}
      </main>

      {/* Footer */}
      <footer className="border-t border-slate-900 bg-slate-950 py-4 text-center text-xs text-slate-500">
        <p>
          {schoolSettings?.schoolName || 'Sistem Kehadiran Sekolah'} &copy; {new Date().getFullYear()}
          {' · '}系统由校长统筹管理 · 配合 DELIMa 账号实名打卡
        </p>
      </footer>

    </div>
  );
}

// Helper: Verify caller Firebase Auth ID Token using Firebase Admin SDK
async function verifyCallerAuth(req: Request): Promise<{ uid: string; email?: string } | null> {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return null;
  }
  const idToken = authHeader.split('Bearer ')[1].trim();
  if (!idToken) return null;

  try {
    const decoded = await adminAuth.verifyIdToken(idToken);
    return {
      uid: decoded.uid,
      email: decoded.email,
    };
  } catch (err) {
    console.error('Admin Auth verifyIdToken error:', err);
    return null;
  }
}

// Helper: Check if caller is registered Principal AND user profile exists with active === true
async function isCallerPrincipal(uid: string): Promise<boolean> {
  try {
    const [adminDoc, userDoc] = await Promise.all([
      adminDb.doc(`admins/${uid}`).get(),
      adminDb.doc(`users/${uid}`).get(),
    ]);
    if (!adminDoc.exists || !userDoc.exists) {
      return false;
    }
    const userData = userDoc.data();
    if (!userData || userData.active !== true) {
      console.warn(`Principal account ${uid} is deactivated or invalid.`);
      return false;
    }
    return true;
  } catch (err) {
    console.error('Error verifying principal role and active status:', err);
    return false;
  }
}

// In-memory mutex per user to serialize concurrent requests and avoid double clicks
const userLocks = new Map<string, Promise<any>>();

async function withUserLock<T>(uid: string, fn: () => Promise<T>): Promise<T> {
  const prevLock = userLocks.get(uid) || Promise.resolve();
  let release: () => void;
  const currentLock = new Promise<void>((resolve) => {
    release = resolve;
  });
  userLocks.set(uid, prevLock.then(() => currentLock).catch(() => currentLock));
  await prevLock;
  try {
    return await fn();
  } finally {
    release!();
    if (userLocks.get(uid) === currentLock) {
      userLocks.delete(uid);
    }
  }
}

  // ================= 1. SAFE SCHOOL SETUP (SETUP_TOKEN + TRANSACTION + ROLLBACK) =================
  app.post('/api/setup/init-school', async (req: Request, res: Response) => {
    const { setupToken, schoolName, schoolCode, principalEmail, principalPassword, principalName } = req.body;

    // 1. Strict Server Environment Token Verification (No default fallback key!)
    const expectedToken = process.env.SETUP_TOKEN;
    if (!expectedToken || !expectedToken.trim()) {
      return res.status(500).json({
        error: '系统安全拒绝：服务器尚未配置 SETUP_TOKEN 环境变量。请在服务器配置有效密钥后再执行学校初始化。',
      });
    }
    if (!setupToken || setupToken.trim() !== expectedToken.trim()) {
      return res.status(401).json({
        error: '安全校验失败：无效或缺少一次性初始化设立密钥 (SETUP_TOKEN)。',
      });
    }

    if (!schoolName || !principalEmail || !principalPassword) {
      return res.status(400).json({ error: '请完整填写学校全称、校长邮箱与管理密码。' });
    }
    if (principalPassword.length < 6) {
      return res.status(400).json({ error: '密码最少需 6 位字符。' });
    }

    const cleanEmail = principalEmail.trim().toLowerCase();
    let createdUid: string | null = null;
    let uid = (req.body.principalUid || '').trim();

    try {
      // 2. Pre-check: Ensure school is not already initialized
      const generalDoc = await adminDb.doc('school_settings/general').get();
      if (generalDoc.exists) {
        return res.status(400).json({ error: '系统已完成初始化设定，禁止重复执行！' });
      }

      // 3. Create or resolve Principal account via Admin Auth if UID not provided
      if (!uid) {
        try {
          const createdAuthUser = await adminAuth.createUser({
            email: cleanEmail,
            password: principalPassword,
            displayName: (principalName || '').trim() || '校长',
          });
          createdUid = createdAuthUser.uid;
          uid = createdAuthUser.uid;
        } catch (authErr: any) {
          if (authErr.code === 'auth/email-already-exists') {
            try {
              const existingUser = await adminAuth.getUserByEmail(cleanEmail);
              uid = existingUser.uid;
              await adminAuth.updateUser(uid, {
                password: principalPassword,
                displayName: (principalName || '').trim() || '校长',
              });
            } catch (resolveErr: any) {
              return res.status(400).json({
                error: `该校长邮箱已被占用，且无法重置: ${resolveErr.message}`,
              });
            }
          } else {
            return res.status(400).json({
              error: `创建校长认证账号失败: ${authErr.message || '邮箱可能已被注册'}`,
            });
          }
        }
      }

      // 4. Atomic Firestore Transaction for School Settings, Admin Registry, and Profile
      await adminDb.runTransaction(async (transaction) => {
        const settingsRef = adminDb.doc('school_settings/general');
        const settingsSnap = await transaction.get(settingsRef);
        if (settingsSnap.exists) {
          throw new Error('ALREADY_INITIALIZED');
        }

        const adminRef = adminDb.doc(`admins/${uid}`);
        const userRef = adminDb.doc(`users/${uid}`);

        transaction.set(settingsRef, {
          schoolName: schoolName.trim(),
          schoolCode: (schoolCode || '').trim().toUpperCase(),
          principalEmail: cleanEmail,
          principalUid: uid,
          createdAt: FieldValue.serverTimestamp(),
          updatedAt: FieldValue.serverTimestamp(),
        });

        transaction.set(adminRef, {
          email: cleanEmail,
          role: 'PRINCIPAL',
          createdAt: FieldValue.serverTimestamp(),
        });

        transaction.set(userRef, {
          uid,
          name: (principalName || '').trim() || '校长',
          email: cleanEmail,
          category: 'PRINCIPAL',
          role: 'PRINCIPAL',
          hasClerkAccess: true,
          active: true,
          createdAt: FieldValue.serverTimestamp(),
          updatedAt: FieldValue.serverTimestamp(),
        });
      });

      return res.json({ success: true, uid });
    } catch (err: any) {
      console.error('Error during school setup transaction:', err);

      // Recovery rollback: Delete created auth user if database transaction failed
      if (createdUid) {
        try {
          await adminAuth.deleteUser(createdUid);
          console.log(`Rolled back auth user ${createdUid} after transaction failure.`);
        } catch (cleanupErr) {
          console.error('Failed to cleanup auth user on rollback:', cleanupErr);
        }
      }

      const msg = err.message === 'ALREADY_INITIALIZED'
        ? '系统已完成初始化设立，禁止重复执行！'
        : (err.message || '初始化数据库事务失败，已回滚。');

      try {
        const result = await adminDb.runTransaction(async (transaction) => {
          // 1. Verify user profile exists and active === true (missing active also rejected)
          const userRef = adminDb.doc(`users/${caller.uid}`);
          const userDoc = await transaction.get(userRef);
          if (!userDoc.exists) {
            throw new Error('UNAUTHORIZED: 您的账号未获校方授权建档，无法打卡。');
          }
          const userData = userDoc.data();
          if (!userData || userData.active !== true) {
            throw new Error('DEACTIVATED: 您的账号已被停用或缺少有效状态，禁止打卡。');
          }

          // 2. Read user attendance state record
          const stateRef = adminDb.doc(`user_attendance_state/${caller.uid}`);
          const stateDoc = await transaction.get(stateRef);
          const state = stateDoc.exists ? stateDoc.data()! : { currentStatus: 'OUT', activeSessionId: null };

          // 3. Check for unclosed session (Requirement 5: Delete auto-close, require real check out!)
          if (state.currentStatus === 'IN' || state.activeSessionId) {
            const sessionDate = state.activeSessionDate || '之前';
            throw new Error(
              `UNCLOSED_SESSION: 您上一组打卡（${sessionDate}）尚未 Check Out。请先点击 Check Out 记录真实离校时间与地点后，才能开始新的一组打卡。`
            );
          }

          // 4. Server-computed Malaysia Date (Asia/Kuala_Lumpur)
          const todayStr = getMalaysiaDateString();
          let slotCountToday = 0;
          if (state.lastDate === todayStr) {
            slotCountToday = state.todaySlotCount || 0;
          }

          // Max 2 slots per day rule
          if (slotCountToday >= 2) {
            throw new Error('MAX_SLOTS_REACHED: 今日已完成两组打卡（每日上限2组），禁止打第三组！');
          }

          const newSlot = slotCountToday + 1; // 1 or 2
          const recordId = `${caller.uid}_${todayStr}_${newSlot}`;

          // 5. Ensure record does not already exist
          const recordRef = adminDb.doc(`attendance/${recordId}`);
          const recordSnap = await transaction.get(recordRef);
          if (recordSnap.exists) {
            throw new Error('RECORD_EXISTS: 该打卡轮次记录已存在，禁止覆盖。');
          }

          const geoRef = adminDb.doc(`attendance_geo/${recordId}`);

          // 6. Atomic Write: Attendance + Isolated Geo + User State
          transaction.set(recordRef, {
            id: recordId,
            userId: caller.uid,
            userName: userData.name || '',
            userEmail: userData.email || caller.email || '',
            category: userData.category || 'GURU',
            date: todayStr,
            slot: newSlot,
            status: 'IN',
            checkInTime: FieldValue.serverTimestamp(),
            createdAt: FieldValue.serverTimestamp(),
            updatedAt: FieldValue.serverTimestamp(),
          });

          transaction.set(geoRef, {
            recordId,
            userId: caller.uid,
            date: todayStr,
            slot: newSlot,
            checkInLat: latitude,
            checkInLng: longitude,
            checkInAccuracy: Math.round(accuracy),
            updatedAt: FieldValue.serverTimestamp(),
          });

          transaction.set(
            stateRef,
            {
              activeSessionId: recordId,
              activeSessionDate: todayStr,
              activeSessionSlot: newSlot,
              lastDate: todayStr,
              todaySlotCount: newSlot,
              currentStatus: 'IN',
            },
            { merge: true }
          );

          return { slot: newSlot, date: todayStr };
        });

        return res.json({
          success: true,
          slot: result.slot,
          date: result.date,
          message: `第 ${result.slot} 轮 Check In 成功！`,
        });
      } catch (err: any) {
        console.error('Punch In Transaction Error:', err);
        const msg = err.message.replace(/^[A-Z_]+:\s*/, '');
        return res.status(400).json({ error: msg || '打卡事务执行失败，请重试。' });
      }
    });
  });

  // ================= 5. ATTENDANCE PUNCH-OUT (FIRESTORE TRANSACTION & ATOMICITY) =================
  app.post('/api/attendance/punch-out', async (req: Request, res: Response) => {
    const caller = await verifyCallerAuth(req);
    if (!caller) {
      return res.status(401).json({ error: '未授权：请先登录系统' });
    }

    const { latitude, longitude, accuracy, earlyCheckOutReason } = req.body;
    if (typeof latitude !== 'number' || typeof longitude !== 'number' || typeof accuracy !== 'number') {
      return res.status(400).json({ error: '未检测到有效 GPS 定位，Check Out 未成功。' });
    }

    return withUserLock(caller.uid, async () => {
      try {
        const result = await adminDb.runTransaction(async (transaction) => {
          // 1. Verify user profile exists and active === true (missing active also rejected)
          const userRef = adminDb.doc(`users/${caller.uid}`);
          const userDoc = await transaction.get(userRef);
          const userData = userDoc.data();
          if (!userDoc.exists || !userData || userData.active !== true) {
            throw new Error('DEACTIVATED: 账号已被停用或无打卡权限（要求 active === true）。');
          }

          // 2. Read user attendance state
          const stateRef = adminDb.doc(`user_attendance_state/${caller.uid}`);
          const stateDoc = await transaction.get(stateRef);
          const state = stateDoc.exists ? stateDoc.data()! : { currentStatus: 'OUT', activeSessionId: null };

          if (state.currentStatus !== 'IN' || !state.activeSessionId) {
            throw new Error('NO_ACTIVE_SESSION: 当前没有进行中的打卡记录，无法 Check Out。');
          }

          const recordRef = adminDb.doc(`attendance/${state.activeSessionId}`);
          const recordSnap = await transaction.get(recordRef);
          if (!recordSnap.exists || recordSnap.data()?.status !== 'IN') {
            throw new Error('SESSION_NOT_FOUND: 打卡记录不存在或状态异常。');
          }

          const geoRef = adminDb.doc(`attendance_geo/${state.activeSessionId}`);

          // 3. Atomic Update: Record completed with REAL server time + Real coordinates (even across days)
          const updatePayload: Record<string, any> = {
            status: 'COMPLETED',
            checkOutTime: FieldValue.serverTimestamp(),
            updatedAt: FieldValue.serverTimestamp(),
          };
          if (earlyCheckOutReason && typeof earlyCheckOutReason === 'string' && earlyCheckOutReason.trim()) {
            updatePayload.earlyCheckOutReason = earlyCheckOutReason.trim();
            updatePayload.isEarlyCheckOut = true;
          }

          transaction.update(recordRef, updatePayload);

          transaction.set(
            geoRef,
            {
              checkOutLat: latitude,
              checkOutLng: longitude,
              checkOutAccuracy: Math.round(accuracy),
              updatedAt: FieldValue.serverTimestamp(),
            },
            { merge: true }
          );

          transaction.set(
            stateRef,
            {
              activeSessionId: null,
              activeSessionDate: null,
              currentStatus: 'OUT',
            },
            { merge: true }
          );

          return { slot: state.activeSessionSlot || 1, date: state.activeSessionDate };
        });

        return res.json({
          success: true,
          slot: result.slot,
          message: `Check Out 成功！已真实记录离校时间与地点。`,
        });
      } catch (err: any) {
        console.error('Punch Out Transaction Error:', err);
        const msg = err.message.replace(/^[A-Z_]+:\s*/, '');
        return res.status(400).json({ error: msg || 'Check Out 事务执行失败，请重试。' });
      }
    });
  });

  // ================= 6. PRINCIPAL STAFF MANAGEMENT & SETTINGS (Backend Verified) =================
  app.post('/api/staff/update-status', async (req: Request, res: Response) => {
    const caller = await verifyCallerAuth(req);
    if (!caller) return res.status(401).json({ error: '未授权：请先登录' });
    const isPrincipal = await isCallerPrincipal(caller.uid);
    if (!isPrincipal) return res.status(403).json({ error: '禁止访问：只有校长可以修改员工档案状态' });

    const { targetUid, active, hasClerkAccess } = req.body;
    if (!targetUid) return res.status(400).json({ error: '缺少 targetUid' });

    try {
      const updates: any = { updatedAt: FieldValue.serverTimestamp() };
      if (typeof active === 'boolean') updates.active = active;
      if (typeof hasClerkAccess === 'boolean') {
        updates.hasClerkAccess = hasClerkAccess;
        const userDoc = await adminDb.doc(`users/${targetUid}`).get();
        const cat = userDoc.data()?.category || 'GURU';
        updates.role = hasClerkAccess ? 'CLERK' : cat;
      }
      await adminDb.doc(`users/${targetUid}`).update(updates);
      return res.json({ success: true });
    } catch (err: any) {
      return res.status(500).json({ error: err.message || '更新失败' });
    }
  });

  app.post('/api/staff/delete', async (req: Request, res: Response) => {
    const caller = await verifyCallerAuth(req);
    if (!caller) return res.status(401).json({ error: '未授权：请先登录' });
    const isPrincipal = await isCallerPrincipal(caller.uid);
    if (!isPrincipal) return res.status(403).json({ error: '禁止访问：只有校长可以删除员工档案' });

    const { targetUid } = req.body;
    if (!targetUid) return res.status(400).json({ error: '缺少 targetUid' });
    if (targetUid === caller.uid) return res.status(400).json({ error: '不能删除校长自身档案' });

    try {
      await adminDb.doc(`users/${targetUid}`).delete();
      try {
        await adminAuth.deleteUser(targetUid);
      } catch (authErr) {
        console.warn('Auth user delete cleanup notice:', authErr);
      }
      return res.json({ success: true });
    } catch (err: any) {
      return res.status(500).json({ error: err.message || '删除失败' });
    }
  });

  app.post('/api/school-settings/update', async (req: Request, res: Response) => {
    const caller = await verifyCallerAuth(req);
    if (!caller) return res.status(401).json({ error: '未授权：请先登录' });
    const isPrincipal = await isCallerPrincipal(caller.uid);
    if (!isPrincipal) return res.status(403).json({ error: '禁止访问：只有校长可以更改学校设定' });

    const { schoolName, schoolCode } = req.body;
    if (!schoolName || !schoolName.trim()) {
      return res.status(400).json({ error: '学校全称不能为空' });
    }

    try {
      await adminDb.doc('school_settings/general').set(
        {
          schoolName: schoolName.trim(),
          schoolCode: (schoolCode || '').trim().toUpperCase(),
          updatedAt: FieldValue.serverTimestamp(),
        },
        { merge: true }
      );
      return res.json({ success: true });
    } catch (err: any) {
      return res.status(500).json({ error: err.message || '保存设定失败' });
    }
  });

  // ================= 6. USER ATTENDANCE STATE STATUS CHECK =================
  app.get('/api/attendance/user-state', async (req: Request, res: Response) => {
    const caller = await verifyCallerAuth(req);
    if (!caller) {
      return res.status(401).json({ error: '未授权：请先登录' });
    }

    try {
      const userDoc = await adminDb.doc(`users/${caller.uid}`).get();
      if (!userDoc.exists || userDoc.data()?.active !== true) {
        return res.status(403).json({ error: '账号未获授权或已被停用' });
      }

      const stateDoc = await adminDb.doc(`user_attendance_state/${caller.uid}`).get();
      const state = stateDoc.exists ? stateDoc.data() : { currentStatus: 'OUT', activeSessionId: null };
      return res.json({ state });
    } catch (err: any) {
      return res.status(500).json({ error: err.message });
    }
  });

  // Mount Vite middleware for SPA in development
  if (process.env.NODE_ENV !== 'production') {
    const vite = await createViteServer({
      server: { middlewareMode: true },
      appType: 'spa',
    });
    app.use(vite.middlewares);
  } else {
    app.use(express.static('dist'));
    app.get('*', (req, res) => {
      res.sendFile(path.resolve('dist/index.html'));
    });
  }

  const PORT = Number(process.env.PORT) || 3000;
  app.listen(PORT, '0.0.0.0', () => {
    console.log(`Sistem Kehadiran Sekolah (Admin SDK) server running on port ${PORT}`);
  });
}

startServer();

import React, { useState, useEffect } from 'react';
import { UserProfile, AttendanceRecord, UserAttendanceState } from '../types';
import { auth, db, handleFirestoreError, OperationType } from '../firebase';
import { 
  collection, 
  query, 
  where, 
  onSnapshot, 
  orderBy, 
  limit,
  doc
} from 'firebase/firestore';
import { 
  getTodayDateString, 
  getCurrentMalaysiaTimeString,
  formatTimeFromTimestamp, 
  formatDisplayDate, 
  calculateDuration 
} from '../utils/date';
import { getCurrentGPSPosition, GeolocationError } from '../utils/geo';
import { EarlyCheckoutModal } from './EarlyCheckoutModal';
import { checkGuruEarlyLeave, checkAkpEarlyLeave } from '../utils/checkoutRules';
import { 
  LogIn, 
  LogOut, 
  Clock, 
  Calendar, 
  AlertTriangle, 
  CheckCircle, 
  History, 
  ShieldCheck, 
  Compass,
  Check,
  Info
} from 'lucide-react';

interface StaffPunchInProps {
  currentUser: UserProfile;
}

export const StaffPunchIn: React.FC<StaffPunchInProps> = ({ currentUser }) => {
  const [currentTime, setCurrentTime] = useState<string>('');
  const [currentDateDisplay, setCurrentDateDisplay] = useState<string>('');
  const [todayRecords, setTodayRecords] = useState<AttendanceRecord[]>([]);
  const [historyRecords, setHistoryRecords] = useState<AttendanceRecord[]>([]);
  const [attendanceState, setAttendanceState] = useState<UserAttendanceState | null>(null);
  const [isProcessing, setIsProcessing] = useState<boolean>(false);
  const [feedback, setFeedback] = useState<{
    type: 'success' | 'error';
    title: string;
    message: string;
  } | null>(null);

  // Early checkout modal states
  const [showEarlyCheckoutModal, setShowEarlyCheckoutModal] = useState<boolean>(false);
  const [earlyRuleDescription, setEarlyRuleDescription] = useState<string>('');

  const todayStr = getTodayDateString();

  // Real-time clock update every second strictly formatted in Asia/Kuala_Lumpur
  useEffect(() => {
    const updateTime = () => {
      setCurrentTime(getCurrentMalaysiaTimeString());
      setCurrentDateDisplay(formatDisplayDate(getTodayDateString()));
    };

    updateTime();
    const interval = setInterval(updateTime, 1000);
    return () => clearInterval(interval);
  }, []);

  // 1. Authoritative subscription to user_attendance_state (Direct state tracking)
  useEffect(() => {
    if (!currentUser?.uid) return;

    const unsubscribe = onSnapshot(
      doc(db, 'user_attendance_state', currentUser.uid),
      (docSnap) => {
        if (docSnap.exists()) {
          setAttendanceState(docSnap.data() as UserAttendanceState);
        } else {
          setAttendanceState({ currentStatus: 'OUT', activeSessionId: null });
        }
      },
      (error) => {
        console.warn('user_attendance_state subscription notice:', error);
      }
    );

    return () => unsubscribe();
  }, [currentUser?.uid]);

  // 2. Subscribe to user's today attendance records
  useEffect(() => {
    if (!currentUser?.uid) return;

    const q = query(
      collection(db, 'attendance'),
      where('userId', '==', currentUser.uid),
      where('date', '==', todayStr)
    );

    const unsubscribe = onSnapshot(
      q,
      (snapshot) => {
        const records: AttendanceRecord[] = [];
        snapshot.forEach((docSnap) => {
          records.push(docSnap.data() as AttendanceRecord);
        });
        records.sort((a, b) => a.slot - b.slot);
        setTodayRecords(records);
      },
      (error) => {
        handleFirestoreError(error, OperationType.GET, 'attendance');
      }
    );

    return () => unsubscribe();
  }, [currentUser?.uid, todayStr]);

  // 3. Subscribe to user's recent history (for history view only)
  useEffect(() => {
    if (!currentUser?.uid) return;

    const qHistory = query(
      collection(db, 'attendance'),
      where('userId', '==', currentUser.uid),
      orderBy('date', 'desc'),
      limit(14)
    );

    const unsubscribe = onSnapshot(
      qHistory,
      (snapshot) => {
        const list: AttendanceRecord[] = [];
        snapshot.forEach((docSnap) => {
          list.push(docSnap.data() as AttendanceRecord);
        });
        setHistoryRecords(list);
      },
      (error) => {
        console.warn('History subscription notice:', error);
      }
    );

    return () => unsubscribe();
  }, [currentUser?.uid]);

  // 4. Authoritative slot & unclosed state computation directly from user_attendance_state
  const isCrossDayUnclosed = 
    attendanceState?.currentStatus === 'IN' &&
    !!attendanceState.activeSessionId &&
    !!attendanceState.activeSessionDate &&
    attendanceState.activeSessionDate !== todayStr;

  const slot1 = todayRecords.find((r) => r.slot === 1);
  const slot2 = todayRecords.find((r) => r.slot === 2);
  const unclosedCrossDaySession = isCrossDayUnclosed
    ? {
        date: attendanceState!.activeSessionDate!,
        slot: attendanceState!.activeSessionSlot || 1,
      }
    : null;

  let activeSessionText = '未打卡 / Belum Masuk';
  let canCheckIn = false;
  let canCheckOut = false;
  let currentTargetSlot = 1;
  let isCompletedToday = false;

  if (isCrossDayUnclosed) {
    // Cross-day unclosed session detected directly from user_attendance_state!
    canCheckIn = false;
    canCheckOut = true;
    currentTargetSlot = attendanceState?.activeSessionSlot || 1;
    activeSessionText = `跨日离校待办：您于 ${attendanceState?.activeSessionDate} 第 ${currentTargetSlot} 轮尚未 Check Out。请先真实 Check Out！`;
  } else if (attendanceState?.currentStatus === 'IN') {
    // Active session in progress today
    canCheckIn = false;
    canCheckOut = true;
    currentTargetSlot = attendanceState.activeSessionSlot || 1;
    activeSessionText = `第 ${currentTargetSlot} 轮进行中 (Sesi ${currentTargetSlot} Sedang Bertugas)`;
  } else {
    // OUT status: calculate remaining slots for today
    let slotCountToday = 0;
    if (attendanceState?.lastDate === todayStr && typeof attendanceState.todaySlotCount === 'number') {
      slotCountToday = attendanceState.todaySlotCount;
    } else {
      slotCountToday = todayRecords.filter((r) => r.status === 'COMPLETED').length;
    }

    if (slotCountToday >= 2) {
      canCheckIn = false;
      canCheckOut = false;
      isCompletedToday = true;
      activeSessionText = '今日两组打卡已全数完成 (Maksimum 2 Sesi Selesai)';
    } else if (slotCountToday === 1) {
      canCheckIn = true;
      canCheckOut = false;
      currentTargetSlot = 2;
      activeSessionText = '第一轮已完成，可进行第二轮打卡 (Sesi 2)';
    } else {
      canCheckIn = true;
      canCheckOut = false;
      currentTargetSlot = 1;
      activeSessionText = '准备打卡第一轮 (Sesi 1)';
    }
  }

  // Handle Check In via Secure Server Endpoint
  const handleCheckIn = async () => {
    if (!canCheckIn || isProcessing) return;
    setFeedback(null);
    setIsProcessing(true);

    try {
      // 1. Get high-precision GPS position (Strict verification)
      const location = await getCurrentGPSPosition();

      // 2. Obtain caller ID token
      const idToken = await auth.currentUser?.getIdToken();
      if (!idToken) {
        throw new Error('用户登录凭证已失效，请重新登录。');
      }

      // 3. Post to authoritative backend (Enforces server UTC+8 time, mutex, max 2 slots)
      const res = await fetch('/api/attendance/punch-in', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${idToken}`,
        },
        body: JSON.stringify({
          latitude: location.latitude,
          longitude: location.longitude,
          accuracy: location.accuracy,
        }),
      });

      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.error || '打卡未成功');
      }

      // 4. Show success ONLY after backend confirms
      setFeedback({
        type: 'success',
        title: `Check In 成功！(Berjaya)`,
        message: `第 ${data.slot || currentTargetSlot} 轮上班已登记。伺服器时间与 GPS 校验已保存。\n(Sesi ${data.slot || currentTargetSlot} bermula pada masa pelayan UTC+8.)`,
      });
    } catch (err: any) {
      console.error('Check In error:', err);
      setFeedback({
        type: 'error',
        title: 'Check In 失败！(Gagal)',
        message: err instanceof GeolocationError 
          ? err.message 
          : `打卡未成功：${err.message || '无法连接至云端服务器，请稍候再试。'}\n(Kehadiran belum direkodkan. Sila cuba lagi.)`,
      });
    } finally {
      setIsProcessing(false);
    }
  };

  // Handle Check Out Click with Rules Verification (Guru and AKP)
  const handleCheckOutClick = () => {
    if (!canCheckOut || isProcessing) return;

    // Check early leave rules
    const isGuru = currentUser.category === 'GURU' || currentUser.role === 'PRINCIPAL';
    if (isGuru) {
      const check = checkGuruEarlyLeave(new Date());
      if (check.reasonRequired) {
        setEarlyRuleDescription(check.ruleDescription || '教师提前离校须登记理由。');
        setShowEarlyCheckoutModal(true);
        return;
      }
    } else if (currentUser.category === 'AKP') {
      const activeRecord = todayRecords.find((r) => r.status === 'IN');
      const check = checkAkpEarlyLeave(currentTargetSlot, slot1, activeRecord?.checkInTime, new Date());
      if (check.reasonRequired) {
        setEarlyRuleDescription(check.ruleDescription || 'AKP 净工时不足 7 小时须登记理由。');
        setShowEarlyCheckoutModal(true);
        return;
      }
    }

    // Direct check-out if no reason required
    handleCheckOut();
  };

  // Handle Check Out via Secure Server Endpoint
  const handleCheckOut = async (earlyReason?: string) => {
    if (!canCheckOut || isProcessing) return;
    setFeedback(null);
    setIsProcessing(true);

    try {
      // 1. Get high-precision GPS position
      const location = await getCurrentGPSPosition();

      // 2. Obtain caller ID token
      const idToken = await auth.currentUser?.getIdToken();
      if (!idToken) {
        throw new Error('用户登录凭证已失效，请重新登录。');
      }

      // 3. Post to authoritative backend
      const res = await fetch('/api/attendance/punch-out', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${idToken}`,
        },
        body: JSON.stringify({
          latitude: location.latitude,
          longitude: location.longitude,
          accuracy: location.accuracy,
          earlyCheckOutReason: earlyReason || undefined,
        }),
      });

      const data = await res.json();
      if (!res.ok) {
        throw new Error(data.error || 'Check Out 失败');
      }

      // 4. Show success
      setFeedback({
        type: 'success',
        title: `Check Out 成功！(Berjaya)`,
        message: earlyReason 
          ? `离校已成功登记（已记录理由：${earlyReason}）。辛苦了！\n(Sesi tamat. Alasan telah direkodkan.)`
          : `离校已成功登记。辛苦了！\n(Sesi tamat. Rekod telah disimpan pada masa pelayan.)`,
      });
    } catch (err: any) {
      console.error('Check Out error:', err);
      setFeedback({
        type: 'error',
        title: 'Check Out 失败！(Gagal)',
        message: err instanceof GeolocationError 
          ? err.message 
          : `打卡未成功：${err.message || '数据保存失败，请检查网络后再试。'}\n(Kehadiran belum direkodkan.)`,
      });
    } finally {
      setIsProcessing(false);
    }
  };

  return (
    <div className="max-w-3xl mx-auto px-4 py-6 sm:py-8 space-y-6">
      
      {/* Welcome & Live Time Card */}
      <div className="bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900 border border-slate-700/80 rounded-3xl p-6 sm:p-8 text-white shadow-xl relative overflow-hidden">
        {/* Subtle background glow */}
        <div className="absolute -right-16 -top-16 w-56 h-56 bg-blue-500/10 rounded-full blur-3xl pointer-events-none"></div>

        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
          <div>
            <div className="flex items-center space-x-2 mb-1">
              <span className={`px-2.5 py-0.5 rounded-full text-xs font-semibold uppercase tracking-wider ${
                currentUser.category === 'GURU'
                  ? 'bg-blue-500/20 text-blue-300 border border-blue-500/30'
                  : 'bg-emerald-500/20 text-emerald-300 border border-emerald-500/30'
              }`}>
                {currentUser.category === 'GURU' ? '教师 / GURU' : '职员 / AKP'}
              </span>
              <span className="text-xs text-slate-400">ID: {currentUser.email.split('@')[0]}</span>
            </div>
            <h2 className="text-xl sm:text-2xl font-bold text-white tracking-tight">
              {currentUser.name} 老师/同仁
            </h2>
            <div className="flex items-center text-xs text-slate-300 mt-1 space-x-1.5">
              <Calendar className="w-3.5 h-3.5 text-blue-400" />
              <span>{currentDateDisplay || todayStr}</span>
            </div>
          </div>

          {/* Big Live Digital Clock */}
          <div className="bg-slate-950/70 border border-slate-700/60 rounded-2xl px-5 py-3 text-center sm:text-right shrink-0">
            <div className="text-3xl sm:text-4xl font-mono font-bold text-white tracking-wider">
              {currentTime || '--:--:--'}
            </div>
            <div className="text-[11px] text-emerald-400 font-medium flex items-center justify-center sm:justify-end gap-1 mt-0.5">
              <span className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-ping"></span>
              <span>马来西亚标准时间 (Waktu Piawai Malaysia)</span>
            </div>
          </div>
        </div>

        {/* Status Banner */}
        <div className="mt-5 pt-4 border-t border-slate-700/60 flex items-center justify-between text-xs sm:text-sm">
          <div className="flex items-center space-x-2 text-slate-300">
            <span className="text-slate-400">当前状态:</span>
            <span className="font-medium text-blue-300">{activeSessionText}</span>
          </div>
          <div className="flex items-center space-x-1 text-slate-400 text-xs">
            <Compass className="w-3.5 h-3.5 text-indigo-400" />
            <span>GPS 经纬度即时验证</span>
          </div>
        </div>
      </div>

      {/* Feedback Message Alert */}
      {feedback && (
        <div
          className={`p-4 rounded-2xl border text-sm flex items-start space-x-3 shadow-lg transition-all animate-in fade-in slide-in-from-top-2 ${
            feedback.type === 'success'
              ? 'bg-emerald-950/80 border-emerald-700 text-emerald-100'
              : 'bg-rose-950/80 border-rose-700 text-rose-100'
          }`}
        >
          {feedback.type === 'success' ? (
            <CheckCircle className="w-5 h-5 text-emerald-400 shrink-0 mt-0.5" />
          ) : (
            <AlertTriangle className="w-5 h-5 text-rose-400 shrink-0 mt-0.5" />

interface MonthlyAnalysisViewProps {
  currentUser: UserProfile;
  schoolSettings: SchoolSettings | null;
  staffList: UserProfile[];
  attendanceRecords: AttendanceRecord[];
  customCalendarDays: SchoolCalendarDay[];
  absenceRecords: StaffAbsenceRecord[];
}

export const MonthlyAnalysisView: React.FC<MonthlyAnalysisViewProps> = ({
  currentUser,
  schoolSettings,
  staffList,
  attendanceRecords,
  customCalendarDays,
  absenceRecords,
}) => {
  // Current month string in Malaysia (YYYY-MM)
  const defaultMonthStr = useMemo(() => {
    const d = new Date();
    const y = d.getFullYear();
    const m = (d.getMonth() + 1).toString().padStart(2, '0');
    return `${y}-${m}`;
  }, []);

  const [selectedMonth, setSelectedMonth] = useState<string>(defaultMonthStr);
  const [selectedCategory, setSelectedCategory] = useState<'ALL' | 'GURU' | 'AKP'>('ALL');
  const [selectedUserId, setSelectedUserId] = useState<string>('ALL');
  const [expandedStaffUid, setExpandedStaffUid] = useState<string | null>(null);

  // Modals state
  const [showPrintReport, setShowPrintReport] = useState<boolean>(false);
  const [showCalendarModal, setShowCalendarModal] = useState<boolean>(false);

  // Month navigation helpers
  const handlePrevMonth = () => {
    const [yStr, mStr] = selectedMonth.split('-');
    let y = parseInt(yStr, 10);
    let m = parseInt(mStr, 10) - 1;
    if (m < 1) {
      m = 12;
      y -= 1;
    }
    setSelectedMonth(`${y}-${m.toString().padStart(2, '0')}`);
  };

  const handleNextMonth = () => {
    const [yStr, mStr] = selectedMonth.split('-');
    let y = parseInt(yStr, 10);
    let m = parseInt(mStr, 10) + 1;
    if (m > 12) {
      m = 1;
      y += 1;
    }
    setSelectedMonth(`${y}-${m.toString().padStart(2, '0')}`);
  };

  // Run Monthly Analysis calculation
  const overall = useMemo(() => {
    return calculateMonthlyAnalysis({
      monthStr: selectedMonth,
      staffList,
      attendanceRecords,
      customCalendarDays,
      absenceRecords,
      filterCategory: selectedCategory,
      filterUserId: selectedUserId,
    });
  }, [selectedMonth, staffList, attendanceRecords, customCalendarDays, absenceRecords, selectedCategory, selectedUserId]);

  // CSV Export
  const handleExportCSV = () => {
    const csvData = generateMonthlyAnalysisCSV(overall);
    const blob = new Blob([csvData], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.setAttribute('download', `Analisis_Kehadiran_${overall.monthStr}_${selectedCategory}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
    URL.revokeObjectURL(url);
  };

  // Selected single staff for day-by-day detail
  const singleStaffDetail = useMemo(() => {
    if (selectedUserId !== 'ALL') {
      return overall.staffMetrics.find((s) => s.user.uid === selectedUserId) || null;
    }
    if (expandedStaffUid) {
      return overall.staffMetrics.find((s) => s.user.uid === expandedStaffUid) || null;
    }
    return null;
  }, [selectedUserId, expandedStaffUid, overall.staffMetrics]);

  return (
    <div className="space-y-6">
      
      {/* Top Banner & Control Bar */}
      <div className="bg-slate-900 border border-slate-800 rounded-3xl p-6 shadow-xl">
        <div className="flex flex-col lg:flex-row lg:items-center lg:justify-between gap-4">
          
          <div>
            <div className="flex items-center space-x-2">
              <span className="px-2.5 py-0.5 rounded-full text-xs font-semibold bg-purple-500/20 text-purple-300 border border-purple-500/30">
                月度出勤综合分析 (Analisis Kehadiran Bulanan)
              </span>
              <span className="text-xs text-slate-400">仅校长特权查看</span>
            </div>
            <h2 className="text-xl sm:text-2xl font-bold text-white mt-1">
              {overall.monthLabelZh} ({overall.monthLabelMs}) 出勤统计
            </h2>
            <p className="text-xs text-slate-400 mt-0.5">
              Guru 07:00 / AKP 08:00 迟到界限核验、同分并列排名、未到校原因占比及标准报告列印
            </p>

                  <th className="py-2.5 px-3">出勤状态判定</th>
                  <th className="py-2.5 px-3">最早到校</th>
                  <th className="py-2.5 px-3">备注 / 事由</th>
                </tr>

              <tbody className="divide-y divide-slate-800 font-mono">
                {Object.values(singleStaffDetail.dailyMap).map((d) => (

            </table>
          </div>
        </div>
      )}

          onClose={() => setShowPrintReport(false)}

      {showCalendarModal && (
        <CalendarAbsenceModal
          isOpen={showCalendarModal}
          onClose={() => setShowCalendarModal(false)}
          staffList={staffList}
          currentMonthStr={selectedMonth}
          calendarDays={customCalendarDays}
          absenceRecords={absenceRecords}
        />
      )}

    </div>
  );
};

import React, { useState } from 'react';
import { UserProfile, SchoolCalendarDay, StaffAbsenceRecord, AbsenceCategoryType } from '../types';
import { db } from '../firebase';
import { doc, setDoc, deleteDoc } from 'firebase/firestore';
import { ABSENCE_REASON_MAP, DEFAULT_SARAWAK_HOLIDAYS_2026 } from '../utils/monthlyAnalysis';
import { Calendar, UserX, Plus, Trash2, X, AlertCircle, CheckCircle, ShieldCheck } from 'lucide-react';

interface CalendarAbsenceModalProps {
  isOpen: boolean;
  onClose: () => void;
  staffList: UserProfile[];
  currentMonthStr: string; // YYYY-MM
  calendarDays: SchoolCalendarDay[];
  absenceRecords: StaffAbsenceRecord[];
  onRefreshData?: () => void;
}

export const CalendarAbsenceModal: React.FC<CalendarAbsenceModalProps> = ({
  isOpen,
  onClose,
  staffList,
  currentMonthStr,
  calendarDays,
  absenceRecords,
}) => {
  const [activeSubTab, setActiveSubTab] = useState<'absence' | 'calendar'>('absence');

  // Absence Form state
  const [selectedStaffUid, setSelectedStaffUid] = useState<string>(staffList[0]?.uid || '');
  const [absenceDate, setAbsenceDate] = useState<string>(`${currentMonthStr}-01`);
  const [absenceCategory, setAbsenceCategory] = useState<AbsenceCategoryType>('KURSUS');
  const [absenceReason, setAbsenceReason] = useState<string>('');
  const [isSubmittingAbsence, setIsSubmittingAbsence] = useState<boolean>(false);

  // Calendar Form state
  const [holidayDate, setHolidayDate] = useState<string>(`${currentMonthStr}-01`);
  const [holidayName, setHolidayName] = useState<string>('');
  const [holidayType, setHolidayType] = useState<'PUBLIC_HOLIDAY' | 'SCHOOL_HOLIDAY' | 'SPECIAL_DUTY'>('SCHOOL_HOLIDAY');
  const [isSubmittingHoliday, setIsSubmittingHoliday] = useState<boolean>(false);

  const [feedback, setFeedback] = useState<{ type: 'success' | 'error'; message: string } | null>(null);

  if (!isOpen) return null;

  // Handle Add Absence Record
  const handleAddAbsence = async (e: React.FormEvent) => {
    e.preventDefault();
    setFeedback(null);
    const staff = staffList.find((s) => s.uid === selectedStaffUid);
    if (!staff || !absenceDate || !absenceReason.trim()) {
      setFeedback({ type: 'error', message: '请完整选择教职员、日期并输入事由。' });
      return;
    }

    try {
      setIsSubmittingAbsence(true);
      const isOfficial = ['KURSUS', 'URUSAN_RASMI', 'BENGKEL', 'TAKLIMAT'].includes(absenceCategory);
      const recordId = `${staff.uid}_${absenceDate}`;

      await setDoc(doc(db, 'absence_records', recordId), {
        id: recordId,
        userId: staff.uid,
        userName: staff.name,
        userEmail: staff.email,
        date: absenceDate,
        category: absenceCategory,
        reason: absenceReason.trim(),
        isOfficialDuty: isOfficial,
        createdAt: new Date(),
        updatedAt: new Date(),
      });

      setFeedback({
        type: 'success',
        message: `已成功登记 ${staff.name} 于 ${absenceDate} 的 ${ABSENCE_REASON_MAP[absenceCategory]?.labelZh || absenceCategory} 记录！`,
      });
      setAbsenceReason('');
    } catch (err: any) {
      console.error('Failed to add absence record:', err);
      setFeedback({ type: 'error', message: `登记失败：${err.message || '权限不足或网络异常'}` });
    } finally {
      setIsSubmittingAbsence(false);
    }
  };

  // Handle Delete Absence Record
  const handleDeleteAbsence = async (id: string, name: string, date: string) => {
    if (!confirm(`确定要移除 ${name} 于 ${date} 的未到校记录吗？`)) return;
    try {
      await deleteDoc(doc(db, 'absence_records', id));
      setFeedback({ type: 'success', message: `已移除 ${name} 于 ${date} 的记录。` });
    } catch (err: any) {
      console.error('Delete absence failed:', err);
      setFeedback({ type: 'error', message: `移除失败：${err.message}` });
    }
  };

  // Handle Add School Calendar Holiday
  const handleAddHoliday = async (e: React.FormEvent) => {
    e.preventDefault();
    setFeedback(null);
    if (!holidayDate || !holidayName.trim()) {
      setFeedback({ type: 'error', message: '请完整选择日期并输入假期/活动名称。' });
      return;
    }

    try {
      setIsSubmittingHoliday(true);
      const calId = `cal_${holidayDate}`;
      await setDoc(doc(db, 'school_calendar', calId), {
        date: holidayDate,
        name: holidayName.trim(),
        isSchoolDay: holidayType === 'SPECIAL_DUTY',
        type: holidayType,
      });

      setFeedback({
        type: 'success',
        message: `已成功保存校历设置：${holidayDate} - ${holidayName}！`,
      });
      setHolidayName('');
    } catch (err: any) {
      console.error('Failed to add calendar holiday:', err);
      setFeedback({ type: 'error', message: `保存失败：${err.message}` });
    } finally {
      setIsSubmittingHoliday(false);
    }
  };

  // Handle Delete Calendar Holiday

    } catch (err: any) {

    }

        {/* Close Button */}

          onClick={onClose}
          className="absolute top-5 right-5 p-1.5 text-slate-400 hover:text-white rounded-xl hover:bg-slate-800 transition"
        >
          <X className="w-5 h-5" />
        </button>

        {/* Header */}
        <div className="flex items-center space-x-3 mb-6">
          <div className="w-10 h-10 rounded-2xl bg-indigo-500/20 border border-indigo-500/30 flex items-center justify-center text-indigo-400 shrink-0">
            <Calendar className="w-5 h-5" />
          </div>

          </div>

          >

          <button
            onClick={() => { setActiveSubTab('calendar'); setFeedback(null); }}
            className={`pb-2.5 transition flex items-center gap-1.5 ${

            }`}
          >
            <Calendar className="w-4 h-4" />

        {/* Feedback alert */}
        {feedback && (
          <div className={`p-3 rounded-xl border text-xs mb-4 flex items-center justify-between ${

        {/* ================= TAB 1: ABSENCE REGISTRATION ================= */}
        {activeSubTab === 'absence' && (

                    <option value="PUBLIC_HOLIDAY">公共假期 (无需到校)</option>

/**
 * 迟到标准精确判定（基于 Asia/Kuala_Lumpur 与服务器保存的时间戳）：
 * - GURU：每天 07:00:00 之后首次 Check In 算迟到；刚好 07:00:00 算准时。
 * - AKP：每天 08:00:00 之后首次 Check In 算迟到；刚好 08:00:00 算准时。
 */
export function evaluateLateness(
  timestamp: any,
  effectiveCategory: 'GURU' | 'AKP'
): {
  isLate: boolean;
  timeStr: string;
  hour: number;
  minute: number;
  second: number;
  totalSeconds: number;
  timestampMs: number;
} {
  let date: Date;
  let timestampMs = 0;

  if (timestamp && typeof timestamp.toDate === 'function') {
    date = timestamp.toDate();
    timestampMs = date.getTime();
  } else if (timestamp && timestamp._seconds) {
    timestampMs = timestamp._seconds * 1000 + Math.floor((timestamp._nanoseconds || 0) / 1000000);
    date = new Date(timestampMs);
  } else if (timestamp instanceof Date) {
    date = timestamp;
    timestampMs = date.getTime();
  } else if (typeof timestamp === 'number') {
    timestampMs = timestamp;
    date = new Date(timestampMs);
  } else if (typeof timestamp === 'string') {
    timestampMs = new Date(timestamp).getTime();
    date = new Date(timestampMs);
  } else {
    return {
      isLate: false,
      timeStr: '--:--',
      hour: 0,
      minute: 0,
      second: 0,
      totalSeconds: 0,
      timestampMs: 0,
    };
  }

  // Format parts in Malaysia timezone
  const formatter = new Intl.DateTimeFormat('en-US', {
    timeZone: MALAYSIA_TIMEZONE,
    hour12: false,
    hour: 'numeric',
    minute: 'numeric',
    second: 'numeric',
  });

  const parts = formatter.formatToParts(date);
  let hour = 0;
  let minute = 0;
  let second = 0;

  parts.forEach((p) => {
    if (p.type === 'hour') hour = parseInt(p.value, 10);
    if (p.type === 'minute') minute = parseInt(p.value, 10);
    if (p.type === 'second') second = parseInt(p.value, 10);
  });

  const totalSeconds = hour * 3600 + minute * 60 + second;
  const timeStr = `${hour.toString().padStart(2, '0')}:${minute.toString().padStart(2, '0')}:${second.toString().padStart(2, '0')}`;

  let isLate = false;
  if (effectiveCategory === 'GURU') {
    // 07:00:00 = 7 * 3600 = 25200 seconds. 刚好 07:00:00 算准时，07:00:01 迟到
    isLate = totalSeconds > 25200;
  } else {
    // 08:00:00 = 8 * 3600 = 28800 seconds. 刚好 08:00:00 算准时，08:00:01 迟到
    isLate = totalSeconds > 28800;
  }

  return {
    isLate,
    timeStr,
    hour,
    minute,
    second,
    totalSeconds,
    timestampMs,
  };
}

/**
 * 格式化百分比：分子、分母与两位小数
 * 分母为 0 时显示“不适用 (N/A)”
 */
export function formatRateWithRatio(
  numerator: number,
  denominator: number,
  unit: string = '天'
): { rate: number | null; rateStr: string; ratioStr: string } {
  if (denominator <= 0) {
    return {
      rate: null,
      rateStr: '不适用',
      ratioStr: `(0/0${unit})`,
    };
  }
  const rateVal = Math.min(100, Math.max(0, (numerator / denominator) * 100));
  return {
    rate: rateVal,
    rateStr: `${rateVal.toFixed(2)}%`,
    ratioStr: `(${numerator}/${denominator}${unit})`,
  };
}

/**
 * 并列排名算法（Dense / Standard Tie Ranking）：
 * 成绩相同者同分并列，下一名次顺延（如 1, 1, 3, 4）
 */
export function computeTieRankings<T>(
  list: T[],
  getValue: (item: T) => number | null,
  sortOrder: 'desc' | 'asc'
): Array<{ rank: number; isTie: boolean; staff: T }> {
  // Only rank items with non-null values
  const valid = list
    .map((staff) => ({ staff, val: getValue(staff) }))
    .filter((entry): entry is { staff: T; val: number } => entry.val !== null);

  valid.sort((a, b) => (sortOrder === 'desc' ? b.val - a.val : a.val - b.val));

  const result: Array<{ rank: number; isTie: boolean; staff: T }> = [];

  for (let i = 0; i < valid.length; i++) {
    const currentVal = valid[i].val;
    let rank = 1;
    let isTie = false;

    // Check previous items for tie
    for (let j = 0; j < i; j++) {
      if (Math.abs(valid[j].val - currentVal) < 0.00001) {
        rank = result[j].rank;
        isTie = true;
        result[j].isTie = true;
        break;
      }
    }

    if (!isTie) {
      rank = i + 1;
    }

    // Check if next item ties
    if (i + 1 < valid.length && Math.abs(valid[i + 1].val - currentVal) < 0.00001) {
      isTie = true;
    }

    result.push({
      rank,
      isTie,
      staff: valid[i].staff,
    });
  }

  return result;
}

/**
 * 核心月度出勤计算引擎 (Authoritative Monthly Analysis Engine)
 */
export function calculateMonthlyAnalysis({
  monthStr, // YYYY-MM
  staffList,
  attendanceRecords,
  customCalendarDays = [],
  absenceRecords = [],
  filterCategory = 'ALL', // 'ALL' | 'GURU' | 'AKP'
  filterUserId = 'ALL',
}: {
  monthStr: string;
  staffList: UserProfile[];
  attendanceRecords: AttendanceRecord[];
  customCalendarDays?: SchoolCalendarDay[];
  absenceRecords?: StaffAbsenceRecord[];
  filterCategory?: 'ALL' | 'GURU' | 'AKP';
  filterUserId?: string;
}): MonthlyAnalysisOverall {
  const [yearStr, monthNumStr] = monthStr.split('-');
  const year = parseInt(yearStr, 10);
  const month = parseInt(monthNumStr, 10);

  const monthNamesZh = ['一月', '二月', '三月', '四月', '五月', '六月', '七月', '八月', '九月', '十月', '十一月', '十二月'];
  const monthNamesMs = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember'];

  const monthLabelZh = `${year}年${monthNamesZh[month - 1] || `${month}月`}`;
  const monthLabelMs = `${monthNamesMs[month - 1] || ''} ${year}`;

  // Current date in Malaysia timezone
  const now = new Date();
  const currentTodayFormatter = new Intl.DateTimeFormat('en-CA', {
    timeZone: MALAYSIA_TIMEZONE,
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  });
  const currentTodayStr = currentTodayFormatter.format(now); // YYYY-MM-DD
  const currentYearMonth = currentTodayStr.slice(0, 7);

  const isFutureMonth = monthStr > currentYearMonth;
  const isCurrentMonth = monthStr === currentYearMonth;

  // Days in selected month
  const totalDaysInMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();

  // Determine evaluated dates in this month
  const datesInMonth: Array<{
    dateStr: string;
    dayNum: number;
    dayOfWeek: number; // 0 = Sun, 1 = Mon ... 6 = Sat
    dayZh: string;
    dayMs: string;
    isWeekend: boolean;
    isHoliday: boolean;
    holidayName?: string;
    isFutureDay: boolean;
    isExpectedDutyDay: boolean;
  }> = [];

  const dayNamesZh = ['星期日', '星期一', '星期二', '星期三', '星期四', '星期五', '星期六'];
  const dayNamesMs = ['Ahad', 'Isnin', 'Selasa', 'Rabu', 'Khamis', 'Jumaat', 'Sabtu'];

  // Calendar lookup map
  const calendarMap = new Map<string, SchoolCalendarDay>();
  customCalendarDays.forEach((c) => calendarMap.set(c.date, c));

  let evaluatedDaysCount = 0;

  for (let d = 1; d <= totalDaysInMonth; d++) {
    const dayStr = d.toString().padStart(2, '0');
    const dateStr = `${yearStr}-${monthNumStr}-${dayStr}`;
    const dateObj = new Date(Date.UTC(year, month - 1, d, 12, 0, 0));
    const dayOfWeek = dateObj.getUTCDay();
    const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;

    // Check holiday from custom calendar or Sarawak standard holidays
    let isHoliday = false;
    let holidayName: string | undefined = undefined;

    if (calendarMap.has(dateStr)) {
      const cal = calendarMap.get(dateStr)!;
      if (!cal.isSchoolDay) {
        isHoliday = true;
        holidayName = cal.name;
      }
    } else if (DEFAULT_SARAWAK_HOLIDAYS_2026[dateStr]) {
      isHoliday = true;
      holidayName = DEFAULT_SARAWAK_HOLIDAYS_2026[dateStr];
    }

    // Future day check: "当前月份只统计至今天，未来月份显示暂无统计"
    const isFutureDay = isFutureMonth || (isCurrentMonth && dateStr > currentTodayStr);
    const isExpectedDutyDay = !isWeekend && !isHoliday && !isFutureDay;

    if (!isFutureDay) {
      evaluatedDaysCount++;
    }

    datesInMonth.push({
      dateStr,
      dayNum: d,
      dayOfWeek,
      dayZh: dayNamesZh[dayOfWeek],
      dayMs: dayNamesMs[dayOfWeek],
      isWeekend,
      isHoliday,
      holidayName,
      isFutureDay,
      isExpectedDutyDay,
    });
  }

  // Pre-index attendance records by (userId -> date -> AttendanceRecord[])
  const attendanceByUserAndDate = new Map<string, Map<string, AttendanceRecord[]>>();
  attendanceRecords.forEach((rec) => {
    if (!rec.userId || !rec.date) return;
    if (!attendanceByUserAndDate.has(rec.userId)) {
      attendanceByUserAndDate.set(rec.userId, new Map());
    }
    const userDateMap = attendanceByUserAndDate.get(rec.userId)!;
    const list = userDateMap.get(rec.date) || [];
    list.push(rec);
    userDateMap.set(rec.date, list);
  });

  // Pre-index absence records by (userId -> date -> StaffAbsenceRecord)
  const absenceByUserAndDate = new Map<string, Map<string, StaffAbsenceRecord>>();
  absenceRecords.forEach((abs) => {
    if (!abs.userId || !abs.date) return;
    if (!absenceByUserAndDate.has(abs.userId)) {
      absenceByUserAndDate.set(abs.userId, new Map());
    }
    absenceByUserAndDate.get(abs.userId)!.set(abs.date, abs);
  });

  // Filter staff list according to category and selection
  const candidateStaff = staffList.filter((staff) => {
    const effectiveCat = getEffectiveStaffCategory(staff);
    if (filterCategory === 'GURU' && effectiveCat !== 'GURU') return false;
    if (filterCategory === 'AKP' && effectiveCat !== 'AKP') return false;
    if (filterUserId !== 'ALL' && staff.uid !== filterUserId) return false;
    return true;
  });

  // Step 1: Compute daily check-in info for all staff to determine daily earliest check-in per category
  // Map of dateStr -> category ('GURU' | 'AKP') -> minTimestampMs & array of staffUids
  const earliestByDateAndCat = new Map<string, {
    GURU: { minMs: number; staffUids: string[]; timeStr: string };
    AKP: { minMs: number; staffUids: string[]; timeStr: string };
  }>();

  datesInMonth.forEach(({ dateStr, isFutureDay }) => {
    if (isFutureDay) return;
    earliestByDateAndCat.set(dateStr, {
      GURU: { minMs: Infinity, staffUids: [], timeStr: '' },
      AKP: { minMs: Infinity, staffUids: [], timeStr: '' },
    });
  });

  // Scan attendance records on each date for all staff in candidate/full pool to find true earliest
  staffList.forEach((staff) => {
    const effectiveCat = getEffectiveStaffCategory(staff);
    const userDates = attendanceByUserAndDate.get(staff.uid);
    if (!userDates) return;

    datesInMonth.forEach(({ dateStr, isFutureDay }) => {
      if (isFutureDay) return;
      const dayRecords = userDates.get(dateStr);
      if (!dayRecords || dayRecords.length === 0) return;

      // Take first punch of the day
      const firstPunch = dayRecords.slice().sort((a, b) => (a.slot || 1) - (b.slot || 1))[0];
      if (!firstPunch?.checkInTime) return;

      const evalResult = evaluateLateness(firstPunch.checkInTime, effectiveCat);
      if (evalResult.timestampMs <= 0) return;

      const dayEarliest = earliestByDateAndCat.get(dateStr)?.[effectiveCat];
      if (!dayEarliest) return;

      if (evalResult.timestampMs < dayEarliest.minMs) {
        dayEarliest.minMs = evalResult.timestampMs;
        dayEarliest.staffUids = [staff.uid];
        dayEarliest.timeStr = evalResult.timeStr;
      } else if (evalResult.timestampMs === dayEarliest.minMs) {
        // "同时间并列，同时显示次数及实际时间"
        if (!dayEarliest.staffUids.includes(staff.uid)) {
          dayEarliest.staffUids.push(staff.uid);
        }
      }
    });
  });

  // Step 2: Compute Metrics for each candidate staff member
  const staffMetrics: MonthlyAnalysisStaffMetric[] = candidateStaff.map((staff) => {
    const effectiveCategory = getEffectiveStaffCategory(staff);
    const userDates = attendanceByUserAndDate.get(staff.uid) || new Map<string, AttendanceRecord[]>();
    const userAbsences = absenceByUserAndDate.get(staff.uid) || new Map<string, StaffAbsenceRecord>();

    let expectedDays = 0;
    let actualAttendedDays = 0;
    let lateDays = 0;
    let earliestCount = 0;
    let officialDutyDays = 0;
    let personalLeaveDays = 0;
    let unverifiedDays = 0;
    let extraWeekendDays = 0;

    const earliestRecords: EarliestRecordItem[] = [];
    const reasonBreakdown: Record<string, number> = {};
    const dailyMap: Record<string, DailyAttendanceStatusInfo> = {};

    // Tenure check: check if staff was registered after the start of this month
    let tenureNote: string | undefined = undefined;
    if (staff.createdAt) {
      let createdDateStr = '';
      if (typeof staff.createdAt.toDate === 'function') {
        createdDateStr = currentTodayFormatter.format(staff.createdAt.toDate());
      } else if (staff.createdAt._seconds) {
        createdDateStr = currentTodayFormatter.format(new Date(staff.createdAt._seconds * 1000));
      }
      if (createdDateStr && createdDateStr.slice(0, 7) > monthStr) {
        tenureNote = `该人员于 ${createdDateStr} 创建账号，早于入职期资料待确认`;
      }
    }
    if (!staff.active) {
      tenureNote = tenureNote ? `${tenureNote}；当前账号已停用但保留历史任期` : '当前账号已停用，出勤按该月实际在任记录统计';
    }

    datesInMonth.forEach((dayInfo) => {
      const { dateStr, dayZh, dayMs, isWeekend, isHoliday, holidayName, isFutureDay, isExpectedDutyDay } = dayInfo;

      if (isExpectedDutyDay) {
        expectedDays++;
      }

      const dayRecords = userDates.get(dateStr) || [];
      const hasPunched = dayRecords.length > 0;
      const sortedRecords = dayRecords.slice().sort((a, b) => (a.slot || 1) - (b.slot || 1));
      const slot1Rec = sortedRecords.find((r) => r.slot === 1) || sortedRecords[0];
      const slot2Rec = sortedRecords.find((r) => r.slot === 2);

      let isLate = false;
      let firstCheckInTimeStr: string | undefined = undefined;
      let firstCheckOutTimeStr: string | undefined = undefined;
      let secondCheckInTimeStr: string | undefined = undefined;
      let secondCheckOutTimeStr: string | undefined = undefined;
      let isEarliest = false;

      if (hasPunched && slot1Rec?.checkInTime) {
        const evalRes = evaluateLateness(slot1Rec.checkInTime, effectiveCategory);
        firstCheckInTimeStr = evalRes.timeStr;
        isLate = evalRes.isLate;

        if (slot1Rec.checkOutTime) {
          firstCheckOutTimeStr = evaluateLateness(slot1Rec.checkOutTime, effectiveCategory).timeStr;
        }
        if (slot2Rec?.checkInTime) {
          secondCheckInTimeStr = evaluateLateness(slot2Rec.checkInTime, effectiveCategory).timeStr;
        }
        if (slot2Rec?.checkOutTime) {
          secondCheckOutTimeStr = evaluateLateness(slot2Rec.checkOutTime, effectiveCategory).timeStr;
        }

        // Check if earliest on this day
        const dayEarliest = earliestByDateAndCat.get(dateStr)?.[effectiveCategory];
        if (dayEarliest && dayEarliest.staffUids.includes(staff.uid)) {
          isEarliest = true;
          earliestCount++;
          earliestRecords.push({
            date: dateStr,
            timeStr: evalRes.timeStr,
            timestampMs: evalRes.timestampMs,
          });
        }

        if (isExpectedDutyDay) {
          // Regular expected duty day attended
          actualAttendedDays++;
          if (isLate) {
            lateDays++;
          }
        } else if (!isFutureDay) {
          // Weekend or holiday punch: "非应到校日的额外到校单列，不能令出席率超过100%"
          extraWeekendDays++;
        }
      }

      // Check absence declarations if not punched on expected duty day
      let absenceCat: AbsenceCategoryType | undefined = undefined;
      let absenceReasonText: string | undefined = undefined;

      if (!hasPunched && isExpectedDutyDay) {
        const decl = userAbsences.get(dateStr);
        if (decl) {
          absenceCat = decl.category;
          absenceReasonText = decl.reason;
          if (decl.isOfficialDuty) {
            officialDutyDays++;
          } else {
            personalLeaveDays++;
          }
          reasonBreakdown[decl.category] = (reasonBreakdown[decl.category] || 0) + 1;
        } else {
          // "没有打卡也没有未到校申报的应到校日，列为待核实，不自动认定旷职"
          absenceCat = 'PERLU_PENGESAHAN';
          absenceReasonText = '待核实 (未打卡且未申报)';
          unverifiedDays++;
          reasonBreakdown['PERLU_PENGESAHAN'] = (reasonBreakdown['PERLU_PENGESAHAN'] || 0) + 1;
        }
      }

      // Status label and badge
      let statusLabel = '';
      let statusBadgeClass = '';

      if (isFutureDay) {
        statusLabel = '未来日期';
        statusBadgeClass = 'bg-slate-800 text-slate-500';
      } else if (hasPunched) {
        if (isWeekend || isHoliday) {
          statusLabel = '额外到校 (周末/公休)';
          statusBadgeClass = 'bg-cyan-500/20 text-cyan-300 border border-cyan-500/30';
        } else if (isLate) {
          statusLabel = '出席 (迟到)';
          statusBadgeClass = 'bg-rose-500/20 text-rose-300 border border-rose-500/30';
        } else {
          statusLabel = '准时出席';
          statusBadgeClass = 'bg-emerald-500/20 text-emerald-300 border border-emerald-500/30';
        }
      } else if (isHoliday) {
        statusLabel = holidayName || '学校公休假';
        statusBadgeClass = 'bg-slate-800/80 text-amber-300 border border-amber-500/30';
      } else if (isWeekend) {
        statusLabel = '周末休息';
        statusBadgeClass = 'bg-slate-900 text-slate-500';
      } else if (absenceCat) {
        const meta = ABSENCE_REASON_MAP[absenceCat] || ABSENCE_REASON_MAP['PERLU_PENGESAHAN'];
        statusLabel = meta.labelZh;
        statusBadgeClass = `${meta.color} border`;
      }

      dailyMap[dateStr] = {
        date: dateStr,
        dayZh,
        dayMs,
        isWeekend,
        isHoliday,
        isExpectedDutyDay,
        isFutureDay,
        holidayName,
        hasPunched,
        firstCheckInTimeStr,
        firstCheckOutTimeStr,
        secondCheckInTimeStr,
        secondCheckOutTimeStr,
        isLate,
        isEarliest,
        absenceCategory: absenceCat,
        absenceReason: absenceReasonText,
        statusLabel,
        statusBadgeClass,
      };
    });

    // Best earliest time
    let earliestBestTimeStr = '-';
    if (earliestRecords.length > 0) {
      const sorted = earliestRecords.slice().sort((a, b) => a.timestampMs - b.timestampMs);
      earliestBestTimeStr = sorted[0].timeStr;
    }

    // Attendance rate
    const attFormatted = formatRateWithRatio(actualAttendedDays, expectedDays, '天');
    // Late rate (denominator is actualAttendedDays)
    const lateFormatted = formatRateWithRatio(lateDays, actualAttendedDays, '天');
    // Earliest rate
    const earlyFormatted = formatRateWithRatio(earliestCount, expectedDays, '天');

    return {
      user: staff,
      effectiveCategory,
      expectedDays,
      actualAttendedDays,
      attendanceRate: attFormatted.rate,
      attendanceRateStr: attFormatted.rateStr,
      attendanceRatioStr: attFormatted.ratioStr,
      lateDays,
      lateRate: lateFormatted.rate,
      lateRateStr: lateFormatted.rateStr,
      lateRatioStr: lateFormatted.ratioStr,
      isEverLate: lateDays > 0,
      earliestCount,
      earliestRate: earlyFormatted.rate,
      earliestRateStr: earlyFormatted.rateStr,
      earliestRatioStr: earlyFormatted.ratioStr,
      earliestRecords,
      earliestBestTimeStr,
      officialDutyDays,
      personalLeaveDays,
      unverifiedDays,
      extraWeekendDays,
      reasonBreakdown,
      dailyMap,
      tenureNote,
    };
  });

  // Step 3: Compute Overall Statistics
  // "全校百分比用总人日计算，不能直接平均个人百分比"
  let totalExpectedPersonDays = 0;
  let totalActualAttendedPersonDays = 0;
  let totalLatePersonDays = 0;
  let totalOfficialDutyPersonDays = 0;
  let totalPersonalLeavePersonDays = 0;
  let totalUnverifiedPersonDays = 0;
  let totalExtraWeekendPersonDays = 0;
  let everLateStaffCount = 0;

  const reasonAggregated: Record<string, number> = {};

  staffMetrics.forEach((m) => {
    totalExpectedPersonDays += m.expectedDays;
    totalActualAttendedPersonDays += m.actualAttendedDays;
    totalLatePersonDays += m.lateDays;
    totalOfficialDutyPersonDays += m.officialDutyDays;
    totalPersonalLeavePersonDays += m.personalLeaveDays;
    totalUnverifiedPersonDays += m.unverifiedDays;
    totalExtraWeekendPersonDays += m.extraWeekendDays;

    if (m.isEverLate) {
      everLateStaffCount++;
    }

    Object.entries(m.reasonBreakdown).forEach(([k, count]) => {
      reasonAggregated[k] = (reasonAggregated[k] || 0) + count;
    });
  });

  const totalStaffCount = staffMetrics.length;

  const overallAttFormatted = formatRateWithRatio(totalActualAttendedPersonDays, totalExpectedPersonDays, '人日');
  const overallLateFormatted = formatRateWithRatio(totalLatePersonDays, totalActualAttendedPersonDays, '人日');
  const everLateFormatted = formatRateWithRatio(everLateStaffCount, totalStaffCount, '人');

  // Total declared absence days = official duty + personal leaves (excluding unverified)
  let totalDeclaredAbsencePersonDays = 0;
  Object.entries(reasonAggregated).forEach(([k, days]) => {
    if (k !== 'PERLU_PENGESAHAN') {
      totalDeclaredAbsencePersonDays += days;
    }
  });

  // Build reason summary table
  const reasonCodes = Object.keys(ABSENCE_REASON_MAP);
  const reasonSummary = reasonCodes
    .map((code) => {
      const meta = ABSENCE_REASON_MAP[code];
      const days = reasonAggregated[code] || 0;
      const rateOverExpected = formatRateWithRatio(days, totalExpectedPersonDays, '人日');
      const shareOverDeclared = code !== 'PERLU_PENGESAHAN'
        ? formatRateWithRatio(days, totalDeclaredAbsencePersonDays, '人日')
        : { rate: null, rateStr: '不适用 (未申报)', ratioStr: '-' };

      return {
        code,
        labelZh: meta.labelZh,
        labelMs: meta.labelMs,
        isOfficial: meta.isOfficial,
        days,
        rateOverExpectedStr: rateOverExpected.rateStr,
        rateOverExpectedRatio: rateOverExpected.ratioStr,
        shareOverDeclaredStr: shareOverDeclared.rateStr,
        shareOverDeclaredRatio: shareOverDeclared.ratioStr,
      };
    })
    .filter((r) => r.days > 0 || r.code === 'KURSUS' || r.code === 'CRK' || r.code === 'CUTI_SAKIT' || r.code === 'PERLU_PENGESAHAN');

  // Step 4: Compute Rankings (Highest & Lowest Attendance, Lateness, Earliest Arrival)
  // "最高及最低出席率排名，同分并列"
  const rankingAttendanceHighest = computeTieRankings(
    staffMetrics,
    (s) => s.attendanceRate,
    'desc'
  );

  const rankingAttendanceLowest = computeTieRankings(
    staffMetrics,
    (s) => s.attendanceRate,
    'asc'
  );

  // Lateness ranking: highest lateness rate first
  const rankingLate = computeTieRankings(
    staffMetrics.filter((s) => s.lateDays > 0),
    (s) => s.lateRate,
    'desc'
  );

  // Earliest Arrival ranking: highest earliest arrival count/rate first
  const rankingEarliest = computeTieRankings(
    staffMetrics.filter((s) => s.earliestCount > 0),
    (s) => s.earliestCount,
    'desc'
  );

  return {
    monthStr,
    monthLabelZh,
    monthLabelMs,
    isFutureMonth,
    daysEvaluatedCount: evaluatedDaysCount,
    totalExpectedPersonDays,
    totalActualAttendedPersonDays,
    overallAttendanceRate: overallAttFormatted.rate,
    overallAttendanceRateStr: overallAttFormatted.rateStr,
    overallAttendanceRatioStr: overallAttFormatted.ratioStr,
    totalLatePersonDays,
    overallLateRate: overallLateFormatted.rate,
    overallLateRateStr: overallLateFormatted.rateStr,
    overallLateRatioStr: overallLateFormatted.ratioStr,
    totalStaffCount,
    everLateStaffCount,
    everLateStaffRate: everLateFormatted.rate,
    everLateStaffRateStr: everLateFormatted.rateStr,
    everLateStaffRatioStr: everLateFormatted.ratioStr,
    totalOfficialDutyPersonDays,
    totalPersonalLeavePersonDays,
    totalUnverifiedPersonDays,
    totalExtraWeekendPersonDays,
    totalDeclaredAbsencePersonDays,
    reasonSummary,
    staffMetrics,
    rankingAttendanceHighest,
    rankingAttendanceLowest,
    rankingLate,
    rankingEarliest,
  };
}

export interface EarlyLeaveCheckResult {
  isEarly: boolean;
  reasonRequired: boolean;
  thresholdTimeStr?: string;
  ruleDescription?: string;
  isWeekend?: boolean;
}

/**
 * Helper to get current day-of-week and time in Malaysia (Asia/Kuala_Lumpur)
 */
export function getMalaysiaDateTimeParts(date: Date = new Date()): {
  dayName: 'monday' | 'tuesday' | 'wednesday' | 'thursday' | 'friday' | 'saturday' | 'sunday';
  dayZh: string;
  hour: number;
  minute: number;
  totalMinutes: number;
  timeStr: string;
  isWeekend: boolean;
} {
  const parts = new Intl.DateTimeFormat('en-US', {
    timeZone: MALAYSIA_TIMEZONE,
    weekday: 'long',
    hour: 'numeric',
    minute: 'numeric',
    hour12: false,
  }).formatToParts(date);

  let weekday = '';
  let hourStr = '0';
  let minStr = '0';

  parts.forEach((p) => {
    if (p.type === 'weekday') weekday = p.value.toLowerCase();
    if (p.type === 'hour') hourStr = p.value;
    if (p.type === 'minute') minStr = p.value;
  });

  const hour = parseInt(hourStr, 10);
  const minute = parseInt(minStr, 10);
  const totalMinutes = hour * 60 + minute;
  const timeStr = `${hour.toString().padStart(2, '0')}:${minute.toString().padStart(2, '0')}`;

  const dayMap: Record<string, string> = {
    monday: '星期一',
    tuesday: '星期二',
    wednesday: '星期三',
    thursday: '星期四',
    friday: '星期五',
    saturday: '星期六',
    sunday: '星期日',
  };

  const dayName = (weekday as any) || 'monday';
  const isWeekend = dayName === 'saturday' || dayName === 'sunday';

  return {
    dayName,
    dayZh: dayMap[dayName] || weekday,
    hour,
    minute,
    totalMinutes,
    timeStr,
    isWeekend,
  };
}

/**
 * Guru 提前离校检查（遵守学校最新规定）：
 * 星期一、二：13:00 前离校须填写理由
 * 星期三、四：12:30 前离校须填写理由
 * 星期五：11:30 前离校须填写理由
 * 星期六、日：豁免
 */
export function checkGuruEarlyLeave(now: Date = new Date()): EarlyLeaveCheckResult {
  const { dayName, dayZh, totalMinutes, timeStr, isWeekend } = getMalaysiaDateTimeParts(now);

  if (isWeekend) {
    return {
      isEarly: false,
      reasonRequired: false,
      isWeekend: true,
      ruleDescription: `${dayZh}（周末打卡）豁免提前离校检查。`,
    };
  }

  let thresholdMinutes = 0;
  let thresholdTimeStr = '';

  if (dayName === 'monday' || dayName === 'tuesday') {
    thresholdMinutes = 13 * 60; // 13:00
    thresholdTimeStr = '13:00';
  } else if (dayName === 'wednesday' || dayName === 'thursday') {
    thresholdMinutes = 12 * 60 + 30; // 12:30
    thresholdTimeStr = '12:30';
  } else if (dayName === 'friday') {
    thresholdMinutes = 11 * 60 + 30; // 11:30
    thresholdTimeStr = '11:30';
  }

  if (thresholdMinutes > 0 && totalMinutes < thresholdMinutes) {
    return {
      isEarly: true,
      reasonRequired: true,
      thresholdTimeStr,
      ruleDescription: `${dayZh}教师规定在校时间至 ${thresholdTimeStr}。当前时间为 ${timeStr}，属于提前离校，须填写理由。`,
    };
  }

  return {
    isEarly: false,
    reasonRequired: false,
    thresholdTimeStr,
    ruleDescription: '已达到规定离校时间。',
  };
}

/**
 * Extract millisecond timestamp safely from Firestore Timestamp, Date, or object
 */
export function getTimestampMs(ts: any): number | null {
  if (!ts) return null;
  if (typeof ts.toDate === 'function') return ts.toDate().getTime();
  if (ts._seconds) return ts._seconds * 1000;
  if (ts instanceof Date) return ts.getTime();
  if (typeof ts === 'number') return ts;
  if (typeof ts === 'string') {
    const parsed = new Date(ts).getTime();
    return isNaN(parsed) ? null : parsed;
  }
  return null;
}

/**
 * AKP 提前离校检查（遵守学校最新规定）：
 * AKP 合计两组工时，并扣除每天1小时休息；已通过午休打卡排除的时间不会重复扣除。
 * 净工时不足7小时仍须填写理由，周末豁免。
 */
export function checkAkpEarlyLeave(
  currentSlot: number,
  slot1Record?: AttendanceRecord | null,
  activeCheckInTime?: any,
  now: Date = new Date()
): EarlyLeaveCheckResult {
  const { dayZh, isWeekend } = getMalaysiaDateTimeParts(now);

  // 周末豁免
  if (isWeekend) {
    return {
      isEarly: false,
      reasonRequired: false,
      isWeekend: true,
      ruleDescription: `${dayZh}（周末打卡）豁免工时不足检查。`,
    };
  }

  const nowMs = now.getTime();
  let s1Minutes = 0;
  let s2Minutes = 0;
  let lunchGapMinutes = 0;

  if (currentSlot === 1) {
    // Currently checking out slot 1
    const s1InMs = getTimestampMs(slot1Record?.checkInTime) || getTimestampMs(activeCheckInTime) || nowMs;
    s1Minutes = Math.max(0, Math.floor((nowMs - s1InMs) / 60000));
  } else {
    // Currently checking out slot 2
    if (slot1Record) {
      const s1InMs = getTimestampMs(slot1Record.checkInTime) || 0;
      const s1OutMs = getTimestampMs(slot1Record.checkOutTime) || s1InMs;
      s1Minutes = Math.max(0, Math.floor((s1OutMs - s1InMs) / 60000));

      const s2InMs = getTimestampMs(activeCheckInTime) || nowMs;
      s2Minutes = Math.max(0, Math.floor((nowMs - s2InMs) / 60000));

      if (s2InMs > s1OutMs) {
        lunchGapMinutes = Math.floor((s2InMs - s1OutMs) / 60000);
      }
    } else {
      const s2InMs = getTimestampMs(activeCheckInTime) || nowMs;
      s2Minutes = Math.max(0, Math.floor((nowMs - s2InMs) / 60000));
    }
  }

  const rawTotalMinutes = s1Minutes + s2Minutes;
  // 已通过午休打卡排除的时间不会重复扣除；总休息时间扣除上限为 60 分钟 (1小时)
  const remainingDeduction = Math.max(0, 60 - lunchGapMinutes);
  const netWorkMinutes = Math.max(0, rawTotalMinutes - remainingDeduction);

  const REQUIRED_MINUTES = 7 * 60; // 7 hours = 420 minutes

  if (netWorkMinutes < REQUIRED_MINUTES) {
    const netHours = Math.floor(netWorkMinutes / 60);
    const netMins = Math.round(netWorkMinutes % 60);

    return {
      isEarly: true,
      reasonRequired: true,
      thresholdTimeStr: '7小时',
      ruleDescription: `AKP 规定每日净工时为 7 小时（已扣除 1 小时休息）。当前今日累计净工时约为 ${netHours}小时 ${netMins}分钟，未满 7 小时，离校须填写理由。`,
    };
  }

  return {
    isEarly: false,
    reasonRequired: false,
    ruleDescription: '今日净工时已达到 7 小时标准。',
  };
}

/**
 * 计算教职员今日在校总时长与净时长（供报表与表格使用）
 */
export function calculateDailyWorkTime(
  category: StaffCategory | 'PRINCIPAL' | string,
  slot1?: AttendanceRecord | null,
  slot2?: AttendanceRecord | null
): { totalMinutes: number; netMinutes: number; displayText: string; isShortHours: boolean } {
  const s1InMs = getTimestampMs(slot1?.checkInTime);
  const s1OutMs = getTimestampMs(slot1?.checkOutTime);
  const s2InMs = getTimestampMs(slot2?.checkInTime);
  const s2OutMs = getTimestampMs(slot2?.checkOutTime);

  const s1Mins = s1InMs && s1OutMs ? Math.max(0, Math.floor((s1OutMs - s1InMs) / 60000)) : 0;
  const s2Mins = s2InMs && s2OutMs ? Math.max(0, Math.floor((s2OutMs - s2InMs) / 60000)) : 0;
  const rawTotalMinutes = s1Mins + s2Mins;

  if (rawTotalMinutes === 0 && !s1OutMs && !s2OutMs) {
    return {
      totalMinutes: 0,
      netMinutes: 0,
      displayText: '-',
      isShortHours: false,
    };
  }

  if (category === 'AKP') {
    let lunchGapMins = 0;
    if (s1OutMs && s2InMs && s2InMs > s1OutMs) {
      lunchGapMins = Math.floor((s2InMs - s1OutMs) / 60000);
    }
    const remainingDeduction = Math.max(0, 60 - lunchGapMins);
    const netMinutes = Math.max(0, rawTotalMinutes - remainingDeduction);

    const hours = Math.floor(netMinutes / 60);
    const mins = Math.round(netMinutes % 60);
    const isShortHours = netMinutes < 420;

    return {
      totalMinutes: rawTotalMinutes,
      netMinutes,
      displayText: hours > 0 ? `${hours}小时 ${mins}分 (${hours}j ${mins}m)` : `${mins}分 (${mins}m)`,
      isShortHours,
    };
  } else {
    // GURU & PRINCIPAL
    const hours = Math.floor(rawTotalMinutes / 60);
    const mins = rawTotalMinutes % 60;

    return {
      totalMinutes: rawTotalMinutes,
      netMinutes: rawTotalMinutes,
      displayText: hours > 0 ? `${hours}小时 ${mins}分 (${hours}j ${mins}m)` : `${mins}分 (${mins}m)`,
      isShortHours: false,
    };
  }
}

import { LocationResult } from '../types';

export class GeolocationError extends Error {
  code: number;
  constructor(message: string, code: number = 0) {
    super(message);
    this.name = 'GeolocationError';
    this.code = code;
  }
}

/**
 * Retrieves the device's high-accuracy GPS position.
 * Strictly verifies network connectivity and coordinates.
 */
export async function getCurrentGPSPosition(): Promise<LocationResult> {
  // 1. Verify Internet Connection
  if (typeof navigator !== 'undefined' && !navigator.onLine) {
    throw new GeolocationError(
      '检测到当前设备无网络连接，无法打卡！请检查 Wi-Fi 或移动数据网络。\n(Tiada sambungan internet dikesan. Sila sambung ke internet.)',
      100
    );
  }

  // 2. Verify Geolocation API Availability
  if (typeof navigator === 'undefined' || !navigator.geolocation) {
    throw new GeolocationError(
      '您的浏览器或设备不支持 GPS 定位功能，无法完成打卡。\n(Pelayar peranti anda tidak menyokong fungsi GPS.)',
      101
    );
  }

  // 3. Request GPS coordinates with high accuracy
  return new Promise((resolve, reject) => {
    navigator.geolocation.getCurrentPosition(
      (position) => {
        const { latitude, longitude, accuracy } = position.coords;
        // Verify valid coordinates
        if (typeof latitude !== 'number' || typeof longitude !== 'number' || isNaN(latitude) || isNaN(longitude)) {
          reject(
            new GeolocationError(
              '取得的 GPS 坐标无效，打卡未成功。\n(Koordinat GPS tidak sah. Sila cuba lagi.)',
              102
            )
          );
          return;
        }

        resolve({
          latitude: Number(latitude.toFixed(7)),
          longitude: Number(longitude.toFixed(7)),
          accuracy: Math.round(accuracy),
          timestamp: position.timestamp,
        });
      },
      (error) => {
        let msg = '定位失败，无法打卡。';
        switch (error.code) {
          case error.PERMISSION_DENIED:
            msg =
              '定位失败：已拒绝位置访问权限！\n打卡必须取得当前地理位置。请在手机浏览器（Safari/Chrome）网站设定中将“位置权限”设为“允许”。\n(Kebenaran lokasi ditolak. Sila benarkan akses lokasi GPS pada pelayar telefon anda.)';
            break;
          case error.POSITION_UNAVAILABLE:
            msg =
              '定位失败：目前无法检测到 GPS 信号。请移步至室外或窗边并确保开启手机定位服务。\n(Isyarat GPS tidak dapat dikesan. Sila aktifkan GPS telefon anda.)';
            break;
          case error.TIMEOUT:
            msg =
              '定位失败：获取 GPS 信号超时，未能取得有效坐标。请重试。\n(Masa tamat mendapatkan lokasi GPS. Sila cuba sekali lagi.)';
            break;
          default:
            msg = `定位失败 (${error.message || '未知错误'})，打卡未成功。\n(Ralat lokasi GPS. Kehadiran belum direkodkan.)`;
            break;
        }
        reject(new GeolocationError(msg, error.code));
      },
      {
        enableHighAccuracy: true,
        timeout: 15000,
        maximumAge: 0,
      }
    );
  });
}

/**
 * Format coordinates for Principal view and Google Maps linking
 */
export function formatCoordinates(lat?: number, lng?: number, accuracy?: number): string {
  if (typeof lat !== 'number' || typeof lng !== 'number') return '无坐标数据';
  const acc = typeof accuracy === 'number' ? ` (精度 ±${accuracy}m)` : '';
  return `${lat.toFixed(5)}, ${lng.toFixed(5)}${acc}`;
}

export function getGoogleMapsUrl(lat: number, lng: number): string {
  return `https://www.google.com/maps/search/?api=1&query=${lat},${lng}`;
}

  let activeExec = 0;

  const userLocks = new Map<string, Promise<any>>();

  async function withUserLock<T>(uid: string, fn: () => Promise<T>): Promise<T> {
    const prevLock = userLocks.get(uid) || Promise.resolve();
    let release: () => void;
    const currentLock = new Promise<void>((resolve) => {
      release = resolve;
    });
    userLocks.set(uid, prevLock.then(() => currentLock).catch(() => currentLock));
    await prevLock;
    try {
      return await fn();
    } finally {
      release!();
      if (userLocks.get(uid) === currentLock) {
        userLocks.delete(uid);
      }
    }

    status = 'IN';

  };

    if (status !== 'IN') throw new Error('NO_ACTIVE_SESSION: 当前没有进行中的打卡');
    status = 'OUT';
  };

    performPunchOut();
    // 组2: Check In -> Check Out

  };

      handleCrossDayAction('CHECK_IN');

  });
